\newcommand{\drawmarginguides}{%
  \draw[red,line width=1pt] ([xshift=3cm]current page.south west) -- ([xshift=3cm]current page.north west);
  \draw[blue,line width=1pt] ([xshift=-1.5cm]current page.south east) -- ([xshift=-1.5cm]current page.north east);
  \draw[red,line width=1pt] ([yshift=-2cm]current page.north west) -- ([yshift=-2cm]current page.north east);
  \draw[blue,line width=1pt] ([yshift=2cm]current page.south west) -- ([yshift=2cm]current page.south east);
  \node[anchor=north west,fill=white,inner sep=2pt,text=red,font=\scriptsize]
    at ([xshift=3cm,yshift=-2cm]current page.north west) {left 3cm / top 2cm};
  \node[anchor=south east,fill=white,inner sep=2pt,text=blue,font=\scriptsize]
    at ([xshift=-1.5cm,yshift=2cm]current page.south east) {right 1.5cm / bottom 2cm};
}

\newcommand{\guidedpage}[1]{%
  \null
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south west,inner sep=0pt] at (current page.south west)
      {\includegraphics[page=#1,width=\paperwidth,height=\paperheight]{main.pdf}};
    \drawmarginguides
  \end{tikzpicture}%
  \newpage
}

\begin{document}
\foreach \p in {1,...,81}{\guidedpage{\p}}
\end{document}
